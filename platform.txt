# platform.txt — Ai-Thinker Ai-WB2-12F (BL602) native Arduino build.
#
# Self-contained package: everything the compiler needs ships inside this
# package, so installing it via Boards Manager works with zero extra setup.
#   sdk.path        -> sdk-include/  (SDK C headers, tree structure preserved)
#   toolchain.path  -> tools/riscv-msys  (Windows RISC-V gcc 10.2.0)
#   lib/            -> prebuilt SDK static archives
# make_package.py regenerates sdk-include/ and tools/riscv-msys/ from the
# bl_iot_sdk checkout at package time, so {sdk.path} never points outside
# the package. Do NOT hand-edit sdk.path to an absolute path — it breaks
# Boards Manager installs.
sdk.path={runtime.platform.path}/sdk-include

# ---- toolchain ------------------------------------------------------------
# toolchain.path MUST be a Windows-LOCAL copy of the RISC-V gcc: a gcc driver
# spawned from a UNC path fails to CreateProcess its sub-tools (cc1/cc1plus/as),
# so pointing compiler.path into the SDK's MSYS dir over UNC does not compile.
# install_windows.ps1 provisions tools/riscv-msys from the SDK's
# toolchain\riscv\MSYS (rv32 multilib only).
toolchain.path={runtime.platform.path}/tools/riscv-msys
compiler.path={toolchain.path}/bin/
compiler.c.cmd=riscv64-unknown-elf-gcc
compiler.cpp.cmd=riscv64-unknown-elf-g++
compiler.c.elf.cmd=riscv64-unknown-elf-gcc
compiler.ar.cmd=riscv64-unknown-elf-ar
compiler.objcopy.cmd=riscv64-unknown-elf-objcopy
compiler.size.cmd=riscv64-unknown-elf-size

# core.a is inside --start-group: archives are scanned lazily, so a member is
# only pulled in when a later group pass resolves the symbol that needs it.
recipe.c.combine.pattern="{compiler.path}{compiler.c.elf.cmd}" {compiler.c.elf.flags} {compiler.c.elf.extra_flags} {object_files} -Wl,--start-group "{archive_file_path}" {compiler.sdk.libs} -Wl,--end-group -o "{build.path}/{build.project_name}.elf"

# BL602 flashable image = the ELF minus the ROM/bugkiller sections.
recipe.objcopy.hex.pattern="{compiler.path}{compiler.objcopy.cmd}" -S -O binary -R .romdata -R .rom -R .bugkiller_command -R .bugkiller "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"
